//<script>
(async () => {
  const MODAL = 'kano_keyevent_modal';
  const STYLE = 'kano_keyevent_style';
  const ROOT = '/data/kano_keyevent';
  const CONFIG = `${ROOT}/config.conf`;
  const RULES = `${ROOT}/rules.conf`;
  const KEY_SH = `${ROOT}/key.sh`;
  const SERVICE_SH = `${ROOT}/service.sh`;
  const PID = `${ROOT}/keyevent.pid`;
  const TEST_PID = `${ROOT}/keyevent_test.pid`;
  const LOG = `${ROOT}/keyevent.log`;
  const BOOT = '/sdcard/ufi_tools_boot.sh';
  const BOOT_LINE = `sh ${SERVICE_SH} start`;

  const createRule = (rule = {}) => ({
    id: `${Date.now()}_${Math.random().toString(16).slice(2, 8)}`,
    ENABLED: '1',
    DEV: '/dev/input/event0',
    KEY_FILTER: '',
    MODE: 'single',
    MULTI_COUNT: '2',
    DOUBLE_GAP_MS: '400',
    LONG_PRESS_MS: '2000',
    ACTION_CMD: 'sh /data/kano_softap_manager/service.sh start',
    ...rule
  });

  const state = {
    installed: false,
    running: false,
    testRunning: false,
    boot: false,
    autoLog: false,
    autoLogTimer: null,
    installConfirm: 0,
    installConfirmTimer: null,
    devices: [],
    config: {
      TEST_DEV: '/dev/input/event0',
      DOUBLE_GAP_MS: '400',
      LONG_PRESS_MS: '2000',
      rules: [createRule()]
    }
  };

  const esc = (v) => String(v ?? '')
    .replaceAll('&', '&amp;')
    .replaceAll('<', '&lt;')
    .replaceAll('>', '&gt;')
    .replaceAll('"', '&quot;')
    .replaceAll("'", '&#39;');

  const sq = (v) => `'${String(v ?? '').replace(/'/g, `'\\''`)}'`;
  const shdq = (v) => `"${String(v ?? '')
    .replace(/\\/g, '\\\\')
    .replace(/"/g, '\\"')
    .replace(/\$/g, '\\$')
    .replace(/`/g, '\\`')
    .replace(/\r?\n/g, ' ')}"`;

  const run = async (cmd, timeout = 30000) => runShellWithRoot(cmd, timeout);
  const okRoot = async () => String((await run('whoami'))?.content || '').includes('root');

  const decodeShellValue = (raw) => String(raw || '')
    .replace(/\\`/g, '`')
    .replace(/\\\$/g, '$')
    .replace(/\\"/g, '"')
    .replace(/\\\\/g, '\\');

  const parseAssignments = (text) => {
    const out = {};
    const lines = String(text || '').split('\n');
    for (const lineRaw of lines) {
      const line = lineRaw.trim();
      if (!line || line.startsWith('#') || !line.includes('=')) continue;
      const i = line.indexOf('=');
      const key = line.slice(0, i).trim();
      let val = line.slice(i + 1).trim();
      if (val.startsWith('"') && val.endsWith('"')) {
        val = decodeShellValue(val.slice(1, -1));
      } else if (val.startsWith("'") && val.endsWith("'")) {
        val = val.slice(1, -1);
      }
      out[key] = val;
    }
    return out;
  };

  const parseRuleLine = (line) => {
    if (String(line || '').includes('\t')) {
      const parts = String(line || '').split('\t');
      if (parts.length >= 8) {
        const [ENABLED, DEV, KEY_FILTER, MODE, MULTI_COUNT, DOUBLE_GAP_MS, LONG_PRESS_MS, ...cmdParts] = parts;
        return createRule({ ENABLED, DEV, KEY_FILTER, MODE, MULTI_COUNT, DOUBLE_GAP_MS, LONG_PRESS_MS, ACTION_CMD: cmdParts.join('\t') });
      }
      const [ENABLED, DEV, KEY_FILTER, MODE, MULTI_COUNT, ACTION_CMD] = parts;
      return createRule({ ENABLED, DEV, KEY_FILTER, MODE, MULTI_COUNT, ACTION_CMD });
    }
    const matches = [...String(line || '').matchAll(/([A-Z_]+)=("(?:(?:\\.)|[^"])*"|'[^']*'|[^\s]+)/g)];
    if (!matches.length) return null;
    const rule = createRule();
    matches.forEach((m) => {
      const key = m[1];
      let val = m[2] || '';
      if (val.startsWith('"') && val.endsWith('"')) {
        val = decodeShellValue(val.slice(1, -1));
      } else if (val.startsWith("'") && val.endsWith("'")) {
        val = val.slice(1, -1);
      }
      rule[key] = val;
    });
    return rule;
  };

  const ensureRules = (rules) => rules.map((rule) => createRule(rule));

  const ensureStyle = () => {
    if (document.getElementById(STYLE)) return;
    const s = document.createElement('style');
    s.id = STYLE;
    s.textContent = `
      #${MODAL} .kk-wrap{display:flex;flex-direction:column;gap:12px;font-size:.74rem;}
      #${MODAL} .kk-card{border:1px solid rgba(255,255,255,.08);background:linear-gradient(180deg,rgba(255,255,255,.06),rgba(255,255,255,.03));border-radius:16px;padding:12px;}
      #${MODAL} .kk-grid{display:grid;grid-template-columns:repeat(2,minmax(0,1fr));gap:10px;}
      #${MODAL} .kk-grid3{display:grid;grid-template-columns:repeat(3,minmax(0,1fr));gap:10px;}
      #${MODAL} label{display:flex;flex-direction:column;gap:5px;font-size:.68rem;opacity:.92;}
      #${MODAL} input,#${MODAL} select,#${MODAL} textarea{width:100%;box-sizing:border-box;border-radius:10px;border:1px solid rgba(255,255,255,.12);background:rgba(0,0,0,.18);color:inherit;padding:8px;font-size:.72rem;}
      #${MODAL} textarea{min-height:220px;font-family:monospace;white-space:pre;}
      #${MODAL} .kk-row{display:flex;flex-wrap:wrap;gap:8px;align-items:center;}
      #${MODAL} .kk-rule-list{display:flex;flex-direction:column;gap:10px;}
      #${MODAL} .kk-rule{border:1px solid rgba(255,255,255,.08);background:rgba(0,0,0,.12);border-radius:14px;padding:12px;}
      #${MODAL} .kk-rule-head{display:flex;justify-content:space-between;gap:8px;align-items:center;margin-bottom:10px;}
      #${MODAL} .kk-chip{display:inline-flex;padding:4px 9px;border-radius:999px;background:rgba(255,255,255,.08);font-size:.65rem;font-weight:700;}
      #${MODAL} .kk-ok{color:#8ff0a7;background:rgba(78,214,118,.16);}
      #${MODAL} .kk-bad{color:#ffaaaa;background:rgba(255,80,80,.16);}
      #${MODAL} .kk-warn{color:#ffd08a;background:rgba(255,170,70,.16);}
      #${MODAL} .kk-help{font-size:.65rem;line-height:1.7;opacity:.78;}
      #${MODAL} .kk-empty{padding:14px;border:1px dashed rgba(255,255,255,.14);border-radius:12px;opacity:.76;text-align:center;}
      #${MODAL} .kk-inline-check{display:flex;align-items:center;gap:8px;font-size:.7rem;flex-direction: row;}
      #${MODAL} .kk-inline-check input{width:auto;transform:scale(1.1);}
      @media (max-width:920px){#${MODAL} .kk-grid3{grid-template-columns:1fr 1fr;}}
      @media (max-width:720px){#${MODAL} .kk-grid,#${MODAL} .kk-grid3{grid-template-columns:1fr;}#${MODAL}{width:95%!important;}}
    `;
    document.head.appendChild(s);
  };

  const renderIntoModal = async () => {
    const box = document.querySelector(`#${MODAL} .content`);
    const mod = document.querySelector(`#${MODAL}`);
    if (!box || !mod) return;
    box.innerHTML = render();
    bind(mod);
    syncAutoLogButton();
    await loadLog();
  };

  const stopAutoLog = () => {
    if (state.autoLogTimer) clearInterval(state.autoLogTimer);
    state.autoLogTimer = null;
  };

  const syncAutoLogButton = () => {
    const btn = document.querySelector('#kk_auto_log');
    if (btn) btn.textContent = state.autoLog ? '停止自动刷新' : '开启自动刷新';
  };

  const setAutoLog = (enabled) => {
    state.autoLog = Boolean(enabled && state.installed);
    stopAutoLog();
    if (state.autoLog) {
      state.autoLogTimer = setInterval(() => {
        if (!document.querySelector(`#${MODAL}`) || !state.installed || !state.autoLog) {
          setAutoLog(false);
          return;
        }
        loadLog();
      }, 1000);
    }
    syncAutoLogButton();
  };

  const syncStateFromForm = () => {
    const root = document.querySelector(`#${MODAL}`);
    if (!root) return;
    const testDev = root.querySelector('#kk_test_dev')?.value || state.config.TEST_DEV;
    const gap = root.querySelector('#kk_gap')?.value.trim() || state.config.DOUBLE_GAP_MS;
    const longMs = root.querySelector('#kk_long')?.value.trim() || state.config.LONG_PRESS_MS;
    const rules = [...root.querySelectorAll('.kk-rule')].map((el) => createRule({
      id: el.dataset.ruleId || createRule().id,
      ENABLED: el.querySelector('[data-field="ENABLED"]')?.checked ? '1' : '0',
      DEV: el.querySelector('[data-field="DEV"]')?.value || '/dev/input/event0',
      KEY_FILTER: el.querySelector('[data-field="KEY_FILTER"]')?.value.trim() || '',
      MODE: el.querySelector('[data-field="MODE"]')?.value || 'single',
      MULTI_COUNT: el.querySelector('[data-field="MULTI_COUNT"]')?.value.trim() || '2',
      DOUBLE_GAP_MS: el.querySelector('[data-field="DOUBLE_GAP_MS"]')?.value.trim() || gap,
      LONG_PRESS_MS: el.querySelector('[data-field="LONG_PRESS_MS"]')?.value.trim() || longMs,
      ACTION_CMD: el.querySelector('[data-field="ACTION_CMD"]')?.value.trim() || ''
    }));
    state.config = {
      TEST_DEV: testDev,
      DOUBLE_GAP_MS: gap,
      LONG_PRESS_MS: longMs,
      rules
    };
  };

  const validateConfig = (cfg, options = {}) => {
    const { requireRules = true } = options;
    if (!cfg.TEST_DEV.startsWith('/dev/input/event')) return '测试监听设备不合法';
    if (!/^\d+$/.test(cfg.DOUBLE_GAP_MS) || Number(cfg.DOUBLE_GAP_MS) < 50) return '连击间隔必须是毫秒整数';
    if (!/^\d+$/.test(cfg.LONG_PRESS_MS) || Number(cfg.LONG_PRESS_MS) < 100) return '长按时间必须是毫秒整数';
    if (!requireRules) return '';
    for (let i = 0; i < cfg.rules.length; i++) {
      const rule = cfg.rules[i];
      if (String(rule.ENABLED) !== '1') continue;
      if (!rule.DEV.startsWith('/dev/input/event')) return `规则 ${i + 1} 的监听设备不合法`;
      if (!rule.KEY_FILTER) return `规则 ${i + 1} 未填写 KeyCode`;
      if (!['single', 'multi', 'long'].includes(rule.MODE)) return `规则 ${i + 1} 的触发模式不合法`;
      if (rule.MODE === 'multi' && (!/^\d+$/.test(rule.MULTI_COUNT) || Number(rule.MULTI_COUNT) < 2)) return `规则 ${i + 1} 的连击次数至少为 2`;
      if (!/^\d+$/.test(rule.DOUBLE_GAP_MS) || Number(rule.DOUBLE_GAP_MS) < 50) return `规则 ${i + 1} 的连击间隔必须是毫秒整数`;
      if (!/^\d+$/.test(rule.LONG_PRESS_MS) || Number(rule.LONG_PRESS_MS) < 100) return `规则 ${i + 1} 的长按时间必须是毫秒整数`;
      if (!rule.ACTION_CMD) return `规则 ${i + 1} 未填写触发命令`;
      if (/[\r\n\t]/.test(rule.ACTION_CMD)) return `规则 ${i + 1} 的触发命令不能包含换行或 Tab`;
    }
    return '';
  };

  const buildConfigBody = (cfg) => [
    `TEST_DEV=${shdq(cfg.TEST_DEV)}`,
    `DOUBLE_GAP_MS=${shdq(cfg.DOUBLE_GAP_MS)}`,
    `LONG_PRESS_MS=${shdq(cfg.LONG_PRESS_MS)}`,
    ''
  ].join('\n');

  const ruleField = (value) => String(value ?? '').replace(/[\r\n\t]/g, ' ');
  const buildRulesBody = (cfg) => cfg.rules.map((rule) => [
    ruleField(rule.ENABLED),
    ruleField(rule.DEV),
    ruleField(rule.KEY_FILTER),
    ruleField(rule.MODE),
    ruleField(rule.MULTI_COUNT),
    ruleField(rule.DOUBLE_GAP_MS),
    ruleField(rule.LONG_PRESS_MS),
    ruleField(rule.ACTION_CMD)
  ].join('\t')).join('\n') + (cfg.rules.length ? '\n' : '');

  const saveConfig = async (silent = false, options = {}) => {
    if (!state.installed && !options.allowNotInstalled) {
      if (!silent) createToast('请先安装插件', 'pink');
      return false;
    }
    syncStateFromForm();
    const cfg = {
      ...state.config,
      rules: ensureRules(state.config.rules)
    };
    const err = validateConfig(cfg, options);
    if (err) {
      if (!silent) createToast(err, 'red');
      return false;
    }
    const cfgBody = buildConfigBody(cfg);
    const rulesBody = buildRulesBody(cfg);
    const cmd = `
      mkdir -p ${sq(ROOT)}
      printf %s ${sq(cfgBody)} > ${sq(CONFIG)}
      printf %s ${sq(rulesBody)} > ${sq(RULES)}
      chmod 666 ${sq(CONFIG)} ${sq(RULES)}
    `;
    const r = await run(cmd);
    if (!r?.success) {
      if (!silent) createToast('保存配置失败', 'red');
      return false;
    }
    state.config = cfg;
    if (!silent) createToast('配置已保存', 'green');
    await refresh();
    return true;
  };

  const readStatus = async () => {
    const r = await run(`
check_pid_file() {
  file="$1"
  [ -f "$file" ] || return 1
  while IFS= read -r p || [ -n "$p" ]; do
    [ -n "$p" ] || continue
    kill -0 "$p" 2>/dev/null && return 0
  done < "$file"
  return 1
}
[ -x ${sq(SERVICE_SH)} ] && echo installed=1 || echo installed=0
check_pid_file ${sq(PID)} && echo running=1 || echo running=0
check_pid_file ${sq(TEST_PID)} && echo test_running=1 || echo test_running=0
grep -qxF ${sq(BOOT_LINE)} ${sq(BOOT)} 2>/dev/null && echo boot=1 || echo boot=0
echo __CONFIG__
[ -f ${sq(CONFIG)} ] && timeout 2s awk '{print}' ${sq(CONFIG)}
echo __RULES__
[ -f ${sq(RULES)} ] && timeout 2s awk '{print}' ${sq(RULES)}
`);
    const text = String(r?.content || '');
    state.installed = /(^|\n)installed=1(\n|$)/.test(text);
    state.running = /(^|\n)running=1(\n|$)/.test(text);
    state.testRunning = /(^|\n)test_running=1(\n|$)/.test(text);
    state.boot = /(^|\n)boot=1(\n|$)/.test(text);

    const configPart = text.includes('__CONFIG__') ? text.split('__CONFIG__')[1].split('__RULES__')[0] : '';
    const rulesPart = text.includes('__RULES__') ? text.split('__RULES__')[1] : '';
    const parsedCfg = parseAssignments(configPart);
    const parsedRules = String(rulesPart || '')
      .split('\n')
      .map((line) => line.trim())
      .filter((line) => line && !line.startsWith('#'))
      .map(parseRuleLine)
      .filter(Boolean);

    state.config = {
      TEST_DEV: parsedCfg.TEST_DEV || state.config.TEST_DEV,
      DOUBLE_GAP_MS: parsedCfg.DOUBLE_GAP_MS || state.config.DOUBLE_GAP_MS,
      LONG_PRESS_MS: parsedCfg.LONG_PRESS_MS || state.config.LONG_PRESS_MS,
      rules: ensureRules(parsedRules)
    };
  };

  const loadDevices = async () => {
    const r = await run(`
      for d in /dev/input/event*; do
        [ -e "$d" ] || continue
        n="$(timeout 1s getevent -lp "$d" 2>/dev/null | awk -F'"' '/name:/{print $2; exit}')"
        [ -n "$n" ] || n="unknown"
        printf '%s|%s\n' "$d" "$n"
      done
    `, 12000);
    state.devices = String(r?.content || '')
      .split('\n')
      .map((x) => x.trim())
      .filter(Boolean)
      .map((x) => {
        const [dev, ...rest] = x.split('|');
        return { dev, name: rest.join('|') || 'unknown' };
      });
    const fallbackDev = state.devices[0]?.dev || '/dev/input/event0';
    if (!state.devices.some((x) => x.dev === state.config.TEST_DEV)) state.config.TEST_DEV = fallbackDev;
    state.config.rules = ensureRules(state.config.rules.length ? state.config.rules : []).map((rule) => ({
      ...rule,
      DEV: state.devices.some((x) => x.dev === rule.DEV) ? rule.DEV : fallbackDev
    }));
  };

  const install = async () => {
    if (!(await okRoot())) return createToast('请先启用高级功能/root', 'pink');
    if (state.installed) {
      clearTimeout(state.installConfirmTimer);
      state.installConfirmTimer = setTimeout(() => state.installConfirm = 0, 2500);
      state.installConfirm++;
      if (state.installConfirm < 2) return createToast('已安装。再点一次执行修复/重装', 'pink');
      state.installConfirm = 0;
    }
    const r = await run(`
mkdir -p ${sq(ROOT)}
cat > ${sq(KEY_SH)} <<'KEYEOF'
#!/system/bin/sh

DEV="\${DEV:-/dev/input/event0}"
KEY_FILTER="$1"
MODE="$2"
MULTI_COUNT="$3"
RULE_FILE=""
[ "$MODE" = "group" ] && RULE_FILE="$3"
ACTION_CMD="\${ACTION_CMD:-}"
DOUBLE_GAP_MS="\${DOUBLE_GAP_MS:-400}"
LONG_PRESS_MS="\${LONG_PRESS_MS:-2000}"
LOG_FILE="\${LOG_FILE:-/data/kano_keyevent/keyevent.log}"
PATH="/system/bin:/system/xbin:/vendor/bin:/sbin:$PATH"
TAB="$(printf '\t')"

down_time=0
last_up_time=0
tap_count=0
pending_seq=0

log_line() {
  printf '%s %s\n' "$(date '+%Y-%m-%d %H:%M:%S' 2>/dev/null)" "$1" >> "$LOG_FILE"
}

run_cmd() {
  cmd="$1"
  [ -n "$cmd" ] || return
  sh -c "$cmd" >> "$LOG_FILE" 2>&1 &
}

now_ms() {
  awk '{printf "%d\\n", $1 * 1000}' /proc/uptime
}

sleep_ms() {
  ms="$1"
  delay="$(awk -v ms="$ms" 'BEGIN { if (ms < 1) ms = 1; printf "%.3f", ms / 1000 }' 2>/dev/null)"
  sleep "$delay" 2>/dev/null || sleep 1
}

parse_key_line() {
  found=0
  parsed_key=""
  parsed_state=""
  for token in "$@"; do
    if [ "$found" = "1" ]; then
      parsed_key="$token"
      found=2
      continue
    fi
    if [ "$found" = "2" ]; then
      parsed_state="$token"
      return 0
    fi
    [ "$token" = "EV_KEY" ] && found=1
  done
  return 1
}

group_has_mode() {
  want="$1"
  [ -f "$RULE_FILE" ] || return 1
  while IFS="$TAB" read -r rule_mode rule_count rule_gap rule_long rule_cmd || [ -n "$rule_mode$rule_cmd" ]; do
    [ "$rule_mode" = "$want" ] && return 0
  done < "$RULE_FILE"
  return 1
}

group_max_gap() {
  max="$DOUBLE_GAP_MS"
  [ -f "$RULE_FILE" ] || { echo "$max"; return; }
  while IFS="$TAB" read -r rule_mode rule_count rule_gap rule_long rule_cmd || [ -n "$rule_mode$rule_cmd" ]; do
    case "$rule_mode" in
      single|multi)
        [ -n "$rule_gap" ] || rule_gap="$DOUBLE_GAP_MS"
        if [ "$rule_gap" -gt "$max" ] 2>/dev/null; then max="$rule_gap"; fi
        ;;
    esac
  done < "$RULE_FILE"
  echo "$max"
}

trigger_group_mode() {
  want="$1"
  detail="$2"
  count_filter="$3"
  long_filter="$4"
  [ -f "$RULE_FILE" ] || return 1
  fired=0
  while IFS="$TAB" read -r rule_mode rule_count rule_gap rule_long rule_cmd || [ -n "$rule_mode$rule_cmd" ]; do
    [ "$rule_mode" = "$want" ] || continue
    if [ "$want" = "multi" ] && [ "$rule_count" != "$count_filter" ]; then continue; fi
    if [ "$want" = "long" ] && [ "$rule_long" != "$long_filter" ]; then continue; fi
    run_cmd "$rule_cmd"
    log_line "trigger key=$KEY_FILTER action=$want $detail"
    fired=1
  done < "$RULE_FILE"
  [ "$fired" = "1" ]
}

trigger_best_long() {
  press_time="$1"
  best=0
  [ -f "$RULE_FILE" ] || return 1
  while IFS="$TAB" read -r rule_mode rule_count rule_gap rule_long rule_cmd || [ -n "$rule_mode$rule_cmd" ]; do
    [ "$rule_mode" = "long" ] || continue
    [ -n "$rule_long" ] || rule_long="$LONG_PRESS_MS"
    if [ "$press_time" -ge "$rule_long" ] 2>/dev/null && [ "$rule_long" -gt "$best" ] 2>/dev/null; then
      best="$rule_long"
    fi
  done < "$RULE_FILE"
  [ "$best" -gt 0 ] 2>/dev/null || return 1
  trigger_group_mode "long" "press_time_ms=$press_time threshold_ms=$best" "" "$best"
}

resolve_taps() {
  seq="$1"
  count="$2"
  [ -f "$PENDING_FILE" ] || return
  [ "$(cat "$PENDING_FILE" 2>/dev/null)" = "$seq" ] || return
  if trigger_group_mode "multi" "count=$count" "$count" ""; then
    rm -f "$PENDING_FILE"
    return
  fi
  if [ "$count" = "1" ]; then
    trigger_group_mode "single" "count=1" "" ""
    rm -f "$PENDING_FILE"
  fi
}

schedule_tap_resolution() {
  count="$1"
  gap="$2"
  pending_seq=$((pending_seq + 1))
  seq="$pending_seq"
  printf '%s' "$seq" > "$PENDING_FILE"
  ( sleep_ms "$gap"; resolve_taps "$seq" "$count" ) &
}

handle_group_up() {
  press_time="$1"
  now="$2"
  if trigger_best_long "$press_time"; then
    pending_seq=$((pending_seq + 1))
    printf '%s' "$pending_seq" > "$PENDING_FILE"
    tap_count=0
    last_up_time=0
    return
  fi

  if ! group_has_mode "multi"; then
    group_has_mode "single" && trigger_group_mode "single" "count=1" "" ""
    return
  fi

  gap="$(group_max_gap)"
  if [ "$last_up_time" -eq 0 ] || [ $((now - last_up_time)) -gt "$gap" ]; then
    tap_count=1
  else
    tap_count=$((tap_count + 1))
  fi
  last_up_time="$now"
  schedule_tap_resolution "$tap_count" "$gap"
}

cleanup() {
  [ -n "$GETEVENT_PID" ] && kill "$GETEVENT_PID" 2>/dev/null
  [ -n "$GETEVENT_PID" ] && sleep 1
  [ -n "$GETEVENT_PID" ] && kill -9 "$GETEVENT_PID" 2>/dev/null
  rm -f "$FIFO" "$PENDING_FILE"
}

      [ "$key" != "$KEY_FILTER" ] && continue
      now="$(now_ms)"

      case "$state" in
        DOWN|00000001)
          down_time="$now"

          if [ "$MODE" = "group" ]; then
            handle_group_up "$press_time" "$now"
          else
            case "$MODE" in
              long)
              if [ "$press_time" -ge "$LONG_PRESS_MS" ]; then
                run_cmd "$ACTION_CMD"
                log_line "trigger key=$key action=long press_time_ms=$press_time"
              fi
              ;;
              multi)
              if [ -z "$MULTI_COUNT" ] || [ "$MULTI_COUNT" -lt 2 ] 2>/dev/null; then MULTI_COUNT=2; fi
              if [ "$last_up_time" -eq 0 ] || [ $((now - last_up_time)) -gt "$DOUBLE_GAP_MS" ]; then

                tap_count=$((tap_count + 1))
              fi
              last_up_time="$now"
              if [ "$tap_count" -eq "$MULTI_COUNT" ]; then
                run_cmd "$ACTION_CMD"
                log_line "trigger key=$key action=multi count=$tap_count"
                tap_count=0
                last_up_time=0
              fi
              ;;
              single)
              run_cmd "$ACTION_CMD"
              log_line "trigger key=$key action=single"
              ;;
              *)
              log_line "unknown_mode mode=$MODE"
              ;;
            esac
          fi
          ;;
      esac
      ;;
  esac
done < "$FIFO"

load_config() {

  . "$CONFIG"

pid_file_running() {
  file="$1"
  [ -f "$file" ] || return 1
  while IFS= read -r p || [ -n "$p" ]; do
    [ -n "$p" ] || continue
    kill -0 "$p" 2>/dev/null && return 0
  done < "$file"
  return 1
}

stop_pid_file() {
  file="$1"
  [ -f "$file" ] || return 0
  while IFS= read -r p || [ -n "$p" ]; do
    [ -n "$p" ] || continue
    kill "$p" 2>/dev/null
    sleep 1
    kill -9 "$p" 2>/dev/null
  done < "$file"
  rm -f "$file"
}

kill_tree() {
  parent="$1"
  [ -n "$parent" ] || return 0
  for child in $(ps 2>/dev/null | awk -v p="$parent" '$3 == p {print $2}'); do
    kill_tree "$child"
  done
  kill "$parent" 2>/dev/null
  sleep 1
  kill -9 "$parent" 2>/dev/null
}

kill_matching_processes() {
  (ps -ef 2>/dev/null || ps 2>/dev/null) | awk '/\/data\/kano_keyevent\/key.sh|\/data\/kano_keyevent\/service.sh|\/data\/kano_keyevent|keyevent_fifo_|\/dev\/input\/event/ {print $2}' | while IFS= read -r p; do
    [ -n "$p" ] || continue
    [ "$p" = "$$" ] && continue
    [ "$p" = "$PPID" ] && continue
    kill_tree "$p"
  done
}

stop_all_watchers() {
  stop_pid_file "$PID"
  stop_pid_file "$TEST_PID"
  kill_matching_processes
  rm -f "$PID" "$TEST_PID"
  rm -f "$ROOT"/keyevent_group_*.rules "$ROOT"/keyevent_groups.tmp
}

start_service() {
  mkdir -p "$ROOT"
  touch "$LOG"
  log_service "service_start_requested"
  stop_all_watchers
  load_config
  [ -f "$KEY_SH" ] || { log_service "key_script_missing path=$KEY_SH"; echo "key script missing"; return 1; }
  [ -f "$RULES" ] || { log_service "rules_missing path=$RULES"; echo "rules missing"; return 1; }
  export DOUBLE_GAP_MS="\${DOUBLE_GAP_MS:-400}"
  export LONG_PRESS_MS="\${LONG_PRESS_MS:-2000}"
  export LOG_FILE="$LOG"
  DEFAULT_DOUBLE_GAP_MS="$DOUBLE_GAP_MS"
  DEFAULT_LONG_PRESS_MS="$LONG_PRESS_MS"
  TAB="$(printf '\t')"
  count=0
  rule_count=0
  group_count=0
  GROUPS="$ROOT/keyevent_groups.tmp"
  rm -f "$GROUPS" "$ROOT"/keyevent_group_*.rules
  : > "$PID"
  : > "$GROUPS"
  while IFS= read -r line || [ -n "$line" ]; do
    [ -n "$line" ] || continue
    [ "\${line#\#}" != "$line" ] && { log_service "rule_skip comment"; continue; }
    [ "\${line#ENABLED=}" != "$line" ] && { log_service "rule_skip old_format_need_save"; continue; }
    ENABLED="$(printf '%s\n' "$line" | cut -f1)"
    DEV="$(printf '%s\n' "$line" | cut -f2)"
    KEY_FILTER="$(printf '%s\n' "$line" | cut -f3)"
    MODE="$(printf '%s\n' "$line" | cut -f4)"
    MULTI_COUNT="$(printf '%s\n' "$line" | cut -f5)"
    RULE_DOUBLE_GAP_MS="$(printf '%s\n' "$line" | cut -f6)"
    RULE_LONG_PRESS_MS="$(printf '%s\n' "$line" | cut -f7)"
    ACTION_CMD="$(printf '%s\n' "$line" | cut -f8-)"
    if [ -z "$ACTION_CMD" ] && [ -z "$RULE_LONG_PRESS_MS" ]; then
      ACTION_CMD="$RULE_DOUBLE_GAP_MS"
      RULE_DOUBLE_GAP_MS="$DEFAULT_DOUBLE_GAP_MS"
      RULE_LONG_PRESS_MS="$DEFAULT_LONG_PRESS_MS"
    fi
    [ -n "$RULE_DOUBLE_GAP_MS" ] || RULE_DOUBLE_GAP_MS="$DEFAULT_DOUBLE_GAP_MS"
    [ -n "$RULE_LONG_PRESS_MS" ] || RULE_LONG_PRESS_MS="$DEFAULT_LONG_PRESS_MS"
    [ "\${ENABLED:-1}" = "1" ] || { log_service "rule_skip disabled"; continue; }
    [ -n "$DEV" ] || { log_service "rule_skip empty_dev"; continue; }
    [ -n "$KEY_FILTER" ] || { log_service "rule_skip empty_key dev=$DEV"; continue; }
    [ -n "$MODE" ] || { log_service "rule_skip empty_mode key=$KEY_FILTER"; continue; }
    group_file="$(awk -v d="$DEV" -v k="$KEY_FILTER" 'BEGIN{FS="\t"} $1 == d && $2 == k {print $3; exit}' "$GROUPS" 2>/dev/null)"
    if [ -z "$group_file" ]; then
      group_count=$((group_count + 1))
      group_file="$ROOT/keyevent_group_$group_count.rules"
      : > "$group_file"
      printf '%s\t%s\t%s\n' "$DEV" "$KEY_FILTER" "$group_file" >> "$GROUPS"
    fi
    printf '%s\t%s\t%s\t%s\t%s\n' "$MODE" "$MULTI_COUNT" "$RULE_DOUBLE_GAP_MS" "$RULE_LONG_PRESS_MS" "$ACTION_CMD" >> "$group_file"
    log_service "rule_grouped dev=$DEV key=$KEY_FILTER mode=$MODE gap_ms=$RULE_DOUBLE_GAP_MS long_ms=$RULE_LONG_PRESS_MS group=$group_file"
    rule_count=$((rule_count + 1))
  done < "$RULES"
  if [ "$rule_count" -eq 0 ]; then
    rm -f "$PID" "$GROUPS" "$ROOT"/keyevent_group_*.rules
    log_service "service_start_failed no_enabled_rules"
    echo "no enabled rules"
    return 1
  fi
  while IFS="$TAB" read -r GROUP_DEV GROUP_KEY GROUP_FILE || [ -n "$GROUP_DEV$GROUP_KEY$GROUP_FILE" ]; do
    [ -n "$GROUP_DEV" ] || continue
    export DEV="$GROUP_DEV"
    export LOG_FILE="$LOG"
    log_service "group_launch dev=$GROUP_DEV key=$GROUP_KEY rules=$GROUP_FILE"
    sh "$KEY_SH" "$GROUP_KEY" group "$GROUP_FILE" </dev/null >> "$LOG" 2>&1 &
    echo $! >> "$PID"
    count=$((count + 1))
  done < "$GROUPS"
  if [ "$count" -eq 0 ]; then
    rm -f "$PID" "$GROUPS" "$ROOT"/keyevent_group_*.rules
    log_service "service_start_failed no_watchers"
    echo "no watchers"
    return 1
  fi
  sleep 1
  if ! pid_file_running "$PID"; then
    log_service "service_started_but_dead"
    echo "started but stopped"
    return 1
  fi
  log_service "service_started watchers=$count rules=$rule_count"
  echo "started:$count rules:$rule_count"
}

stop_service() {
  stop_all_watchers
  echo "stopped"
}

    await loadDevices();
    const { id, el } = createModal({
      name: MODAL,
      title: '实用按键触发器',
      maxWidth: '980px',
      contentStyle: 'max-height: 72vh;',
      showConfirm: false,
      onClose: () => {
        setAutoLog(false);
        return true;
      },
      content: render()
    });
    bind(el);
    showModal(id);
    await loadLog();
    setAutoLog(state.installed);
    close();
  };

  const mainBtn = document.createElement('button');
  mainBtn.textContent = '按键触发器';
  mainBtn.onclick = openModal;
  document.querySelector('.actions-buttons')?.appendChild(mainBtn);
})();
