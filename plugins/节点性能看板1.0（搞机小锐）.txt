                'oc': '#FF5722',
                'sa': '#795548'
            };
            return regionColors[region] || '#607D8B';
        };

        // 更新选中的服务器
        const updateSelectedServers = () => {
            selectedServers = speedTestServers.filter(server => server.selected);
            const selectedCount = selectedServers.length;
            
            startBtn.disabled = selectedCount === 0;
            startBtn.style.opacity = selectedCount === 0 ? '0.5' : '1';
            
            if (selectedCount > 0) {
                statusEl.textContent = `已选择 ${selectedCount} 个测速节点`;
                statusEl.style.background = 'rgba(76, 175, 80, 0.1)';
                statusEl.style.borderLeftColor = '#4CAF50';
            } else {
                statusEl.textContent = '请选择测速节点，然后点击开始测试';
                statusEl.style.background = 'rgba(33, 150, 243, 0.1)';
                statusEl.style.borderLeftColor = '#2196F3';
            }
        };

        // 快速选择功能
        selectAllCnBtn.addEventListener('click', () => {
            speedTestServers.forEach(server => {
                server.selected = server.region === 'cn';
            });
            updateServerCheckboxes();
        });

        selectAllIntlBtn.addEventListener('click', () => {
            speedTestServers.forEach(server => {
                server.selected = server.region !== 'cn';
            });
            updateServerCheckboxes();
        });

        selectAllBtn.addEventListener('click', () => {
            speedTestServers.forEach(server => server.selected = true);
            updateServerCheckboxes();
        });

        deselectAllBtn.addEventListener('click', () => {
            speedTestServers.forEach(server => server.selected = false);
            updateServerCheckboxes();
        });

        const updateServerCheckboxes = () => {
            serverCheckboxes.forEach(item => {
                item.checkbox.checked = item.server.selected;
                if (item.server.selected) {
                    item.serverItem.style.background = 'rgba(255, 255, 255, 0.1)';
                    item.serverItem.style.borderColor = item.region.color;
                } else {
                    item.serverItem.style.background = 'rgba(255, 255, 255, 0.05)';
                    item.serverItem.style.borderColor = 'transparent';
                }
            });
            updateSelectedServers();
        };

        // 开始测速
        const startSpeedTest = () => {
            if (isTesting || selectedServers.length === 0) return;
            
            isTesting = true;
            window.speedTestIsTesting = true;
            currentTestIndex = 0;
            testResults = [];
            
            // 更新UI状态
            startBtn.disabled = true;
            startBtn.style.opacity = '0.5';
            stopBtn.disabled = false;
            stopBtn.style.opacity = '1';
            progressEl.style.display = 'block';
            resultsEl.innerHTML = '';
            summaryEl.style.display = 'none';
            summaryPlaceholder.style.display = 'block';
            regionalResultsEl.innerHTML = '';
            statusEl.textContent = `正在测试 ${selectedServers.length} 个节点...`;
            statusEl.style.background = 'rgba(33, 150, 243, 0.1)';
            statusEl.style.borderLeftColor = '#2196F3';
            
            // 重置进度条
            progressBarEl.style.width = '0%';
            progressTextEl.textContent = '正在初始化...';
            progressPercentEl.textContent = '0%';
            
            // 创建结果卡片
            createResultCards();
            
            // 开始第一个服务器测试
            testNextServer();
        };

        // 停止测速
        const stopSpeedTest = () => {
            isTesting = false;
            window.speedTestIsTesting = false;
            startBtn.disabled = false;
            startBtn.style.opacity = '1';
            stopBtn.disabled = true;
            stopBtn.style.opacity = '0.5';
            progressTextEl.textContent = '测试已停止';
            progressPercentEl.textContent = '已停止';
            statusEl.textContent = '测试已停止';
            statusEl.style.background = 'rgba(244, 67, 54, 0.1)';
            statusEl.style.borderLeftColor = '#F44336';
        };

        // 测试下一个服务器
        const testNextServer = () => {
            if (!isTesting || currentTestIndex >= selectedServers.length) {
                finishTesting();
                return;
            }
            
            const server = selectedServers[currentTestIndex];
            const progressPercent = Math.round((currentTestIndex / selectedServers.length) * 100);
            
            progressTextEl.textContent = `正在测试 ${server.name} (${currentTestIndex + 1}/${selectedServers.length})`;
            progressPercentEl.textContent = `${progressPercent}%`;
            progressBarEl.style.width = `${progressPercent}%`;
            
            // 模拟测速过程
            simulateSpeedTest(server);
        };

        // 模拟测速过程
        const simulateSpeedTest = (server) => {
            // 获取结果卡片
            const card = document.getElementById(`speedtest-result-${server.id}`);
            const statusEl = card.querySelector('.status');
            
            // 更新状态
            statusEl.textContent = '测试中';
            statusEl.style.background = '#FF9800';
            
            // 模拟不同的网络延迟，根据地区
            const baseDelay = getBaseDelayByRegion(server.region);
            
            // 阶段1: 延迟测试
            setTimeout(() => {
                const ping = Math.floor(Math.random() * 80) + baseDelay;
                document.getElementById(`speedtest-ping-${server.id}`).textContent = `${ping} ms`;
                document.getElementById(`speedtest-ping-${server.id}`).style.color = getPingColor(ping);
                
                // 阶段2: 下载测试
                simulateDownload(server.id, server.region, () => {
                    // 阶段3: 上传测试
                    simulateUpload(server.id, server.region, () => {
                        // 生成最终结果
                        const result = {
                            serverId: server.id,
                            serverName: server.name,
                            serverRegion: server.region,
                            ping: ping,
                            download: getRealisticDownloadSpeed(server.region),
                            upload: getRealisticUploadSpeed(server.region),
                            jitter: Math.floor(Math.random() * 15) + 1
                        };
                        
                        testResults.push(result);
                        
                        // 更新UI
                        statusEl.textContent = '已完成';
                        statusEl.style.background = '#4CAF50';
                        document.getElementById(`speedtest-jitter-${server.id}`).textContent = `${result.jitter} ms`;
                        
                        // 更新进度
                        currentTestIndex++;
                        const progress = (currentTestIndex / selectedServers.length) * 100;
                        progressBarEl.style.width = `${progress}%`;
                        progressPercentEl.textContent = `${Math.round(progress)}%`;
                        
                        // 继续下一个测试
                        testNextServer();
                    });
                });
            }, 500);
        };

        // 根据地区获取基础延迟
        const getBaseDelayByRegion = (region) => {
            const baseDelays = {
                'cn': 10,    // 国内延迟较低
                'hmt': 30,   // 港澳台延迟中等
                'asia': 50,  // 亚洲其他地区
                'na': 120,   // 北美延迟较高
                'eu': 150,   // 欧洲延迟较高
                'oc': 180,   // 大洋洲延迟高
                'sa': 200    // 南美洲延迟最高
            };
            return baseDelays[region] || 100;
        };

        // 获取延迟颜色
        const getPingColor = (ping) => {
            if (ping < 50) return '#4CAF50';  // 绿色，优秀
            if (ping < 100) return '#FF9800'; // 橙色，良好
            if (ping < 200) return '#FF5722'; // 深橙色，一般
            return '#F44336'; // 红色，差
        };

        // 获取真实的下载速度（基于地区）
        const getRealisticDownloadSpeed = (region) => {
            const baseSpeeds = {
                'cn': 800,    // 国内速度较高
                'hmt': 500,   // 港澳台速度中等
                'asia': 300,  // 亚洲其他
                'na': 200,    // 北美
                'eu': 180,    // 欧洲
                'oc': 150,    // 大洋洲
                'sa': 100     // 南美洲
            };
            
            const baseSpeed = baseSpeeds[region] || 200;
            return Math.floor(Math.random() * (baseSpeed * 0.4)) + baseSpeed * 0.6;
        };

        // 获取真实的上传速度（基于地区）
        const getRealisticUploadSpeed = (region) => {
            const baseSpeeds = {
                'cn': 300,    // 国内上传速度
                'hmt': 200,   // 港澳台上传速度
                'asia': 150,  // 亚洲其他
                'na': 100,    // 北美
                'eu': 90,     // 欧洲
                'oc': 80,     // 大洋洲
                'sa': 60      // 南美洲
            };
            
            const baseSpeed = baseSpeeds[region] || 100;
            return Math.floor(Math.random() * (baseSpeed * 0.4)) + baseSpeed * 0.6;
        };

        // 模拟下载过程
        const simulateDownload = (serverId, region, callback) => {
            let progress = 0;
            const targetSpeed = getRealisticDownloadSpeed(region);
            let currentSpeed = 0;
            
            const interval = setInterval(() => {
                progress += 8;
                currentSpeed = Math.min(targetSpeed, Math.floor(targetSpeed * 0.3 + progress * targetSpeed * 0.007));
                const speedElement = document.getElementById(`speedtest-download-${serverId}`);
                speedElement.textContent = `${currentSpeed} Mbps`;
                speedElement.style.color = getSpeedColor(currentSpeed, 'download', region);
                
                if (progress >= 100) {
                    clearInterval(interval);
                    speedElement.textContent = `${targetSpeed} Mbps`;
                    speedElement.style.color = getSpeedColor(targetSpeed, 'download', region);
                    callback();
                }
            }, 100);
        };

        // 模拟上传过程
        const simulateUpload = (serverId, region, callback) => {
            let progress = 0;
            const targetSpeed = getRealisticUploadSpeed(region);
            let currentSpeed = 0;
            
            const interval = setInterval(() => {
                progress += 12;
                currentSpeed = Math.min(targetSpeed, Math.floor(targetSpeed * 0.3 + progress * targetSpeed * 0.005));
                const speedElement = document.getElementById(`speedtest-upload-${serverId}`);
                speedElement.textContent = `${currentSpeed} Mbps`;
                speedElement.style.color = getSpeedColor(currentSpeed, 'upload', region);
                
                if (progress >= 100) {
                    clearInterval(interval);
                    speedElement.textContent = `${targetSpeed} Mbps`;
                    speedElement.style.color = getSpeedColor(targetSpeed, 'upload', region);
                    callback();
                }
            }, 100);
        };

        // 获取速度颜色
        const getSpeedColor = (speed, type, region) => {
            const thresholds = {
                'download': { excellent: 500, good: 200, fair: 50 },
                'upload': { excellent: 200, good: 50, fair: 20 }
            };
            
            const threshold = thresholds[type];
            if (speed >= threshold.excellent) return '#4CAF50';
            if (speed >= threshold.good) return '#2196F3';
            if (speed >= threshold.fair) return '#FF9800';
            return '#F44336';
        };

        // 创建结果卡片
        const createResultCards = () => {
            resultsEl.innerHTML = '';
            
            // 按测试顺序添加卡片
            selectedServers.forEach((server, index) => {
                const regionColor = getRegionColor(server.region);
                
                const cardHTML = `
                    <div id="speedtest-result-${server.id}" style="
                        background: rgba(255, 255, 255, 0.05);
                        padding: 20px;
                        border-radius: 12px;
                        border-left: 4px solid ${regionColor};
                        transition: all 0.3s ease;
                    ">
                        <div style="display:flex; justify-content:space-between; align-items:center; margin-bottom:15px;">
                            <div>
                                <div style="display:flex; align-items:center; gap:10px; margin-bottom:5px;">
                                    <span style="font-weight:bold; font-size:16px; color:white;">${server.name}</span>
                                    <span style="
                                        font-size:11px;
                                        color: white;
                                        padding: 3px 8px;
                                        background: ${regionColor};
                                        border-radius: 12px;
                                    ">${getRegionName(server.region)}</span>
                                </div>
                                <div style="font-size:12px; color:rgba(255, 255, 255, 0.7);">${server.location}</div>
                            </div>
                            <span class="status" style="
                                font-size:12px;
                                padding: 6px 12px;
                                border-radius: 20px;
                                background: #666;
                                color: white;
                                font-weight: bold;
                            ">等待中</span>
                        </div>
                        <div style="
                            display: grid;
                            grid-template-columns: repeat(2, 1fr);
                            gap: 15px;
                        ">
                            <div style="text-align:center; padding:15px; background:rgba(0,0,0,0.2); border-radius:8px;">
                                <div style="font-size:12px; color:rgba(255,255,255,0.7); margin-bottom:8px;">下载速度</div>
                                <div id="speedtest-download-${server.id}" style="font-size:22px; font-weight:bold; color:#4CAF50;">-</div>
                            </div>
                            <div style="text-align:center; padding:15px; background:rgba(0,0,0,0.2); border-radius:8px;">
                                <div style="font-size:12px; color:rgba(255,255,255,0.7); margin-bottom:8px;">上传速度</div>
                                <div id="speedtest-upload-${server.id}" style="font-size:22px; font-weight:bold; color:#2196F3;">-</div>
                            </div>
                            <div style="text-align:center; padding:15px; background:rgba(0,0,0,0.2); border-radius:8px;">
                                <div style="font-size:12px; color:rgba(255,255,255,0.7); margin-bottom:8px;">网络延迟</div>
                                <div id="speedtest-ping-${server.id}" style="font-size:22px; font-weight:bold; color:#FF9800;">-</div>
                            </div>
                            <div style="text-align:center; padding:15px; background:rgba(0,0,0,0.2); border-radius:8px;">
                                <div style="font-size:12px; color:rgba(255,255,255,0.7); margin-bottom:8px;">网络抖动</div>
                                <div id="speedtest-jitter-${server.id}" style="font-size:22px; font-weight:bold; color:#9C27B0;">-</div>
                            </div>
                        </div>
                    </div>
                `;
                resultsEl.insertAdjacentHTML('beforeend', cardHTML);
            });
        };

        // 完成测试
        const finishTesting = () => {
            if (!isTesting) return;
            
            isTesting = false;
            window.speedTestIsTesting = false;
            startBtn.disabled = false;
            startBtn.style.opacity = '1';
            stopBtn.disabled = true;
            stopBtn.style.opacity = '0.5';
            
            progressTextEl.textContent = '所有测试已完成';
            progressBarEl.style.width = '100%';
            progressPercentEl.textContent = '100%';
            statusEl.textContent = `测试完成！共测试了 ${testResults.length} 个节点`;
            statusEl.style.background = 'rgba(76, 175, 80, 0.1)';
            statusEl.style.borderLeftColor = '#4CAF50';
            
            // 显示摘要
            if (testResults.length > 0) {
                const bestDownload = Math.max(...testResults.map(r => r.download));
                const bestUpload = Math.max(...testResults.map(r => r.upload));
                const avgPing = Math.round(testResults.reduce((sum, r) => sum + r.ping, 0) / testResults.length);
                
                document.getElementById('speedtest-best-download').textContent = `${bestDownload} Mbps`;
                document.getElementById('speedtest-best-upload').textContent = `${bestUpload} Mbps`;
                document.getElementById('speedtest-avg-ping').textContent = `${avgPing} ms`;
                summaryEl.style.display = 'block';
                summaryPlaceholder.style.display = 'none';
                
                // 显示各地区最佳结果
                showRegionalResults();
            }
        };

        // 显示各地区最佳结果
        const showRegionalResults = () => {
            // 按地区分组
            const regionalResults = {};
            testResults.forEach(result => {
                if (!regionalResults[result.serverRegion]) {
                    regionalResults[result.serverRegion] = [];
                }
                regionalResults[result.serverRegion].push(result);
            });
            
            let regionalHTML = '<div style="font-size: 14px; color: rgba(255, 255, 255, 0.7); margin-bottom: 15px;">各地区最佳表现：</div>';
            regionalHTML += '<div style="display: grid; grid-template-columns: repeat(auto-fill, minmax(200px, 1fr)); gap: 15px;">';
            
            Object.keys(regionalResults).forEach(region => {
                const results = regionalResults[region];
                if (results.length > 0) {
                    const bestDownload = Math.max(...results.map(r => r.download));
                    const bestUpload = Math.max(...results.map(r => r.upload));
                    const bestPing = Math.min(...results.map(r => r.ping));
                    const regionName = getRegionName(region);
                    const regionColor = getRegionColor(region);
                    
                    regionalHTML += `
                        <div style="
                            padding: 15px;
                            background: rgba(255, 255, 255, 0.05);
                            border-radius: 10px;
                            border-left: 4px solid ${regionColor};
                        ">
                            <div style="font-size: 14px; color: ${regionColor}; font-weight: bold; margin-bottom: 10px;">${regionName}</div>
                            <div style="display: flex; justify-content: space-between; margin-bottom: 5px;">
                                <span style="font-size: 12px; color: rgba(255, 255, 255, 0.7);">下载:</span>
                                <span style="font-size: 14px; color: #4CAF50; font-weight: bold;">${bestDownload}Mbps</span>
                            </div>
                            <div style="display: flex; justify-content: space-between; margin-bottom: 5px;">
                                <span style="font-size: 12px; color: rgba(255, 255, 255, 0.7);">上传:</span>
                                <span style="font-size: 14px; color: #2196F3; font-weight: bold;">${bestUpload}Mbps</span>
                            </div>
                            <div style="display: flex; justify-content: space-between;">
                                <span style="font-size: 12px; color: rgba(255, 255, 255, 0.7);">延迟:</span>
                                <span style="font-size: 14px; color: ${getPingColor(bestPing)}; font-weight: bold;">${bestPing}ms</span>
                            </div>
                        </div>
                    `;
                }
            });
            
            regionalHTML += '</div>';
            regionalResultsEl.innerHTML = regionalHTML;
        };

        // 绑定事件
        startBtn.addEventListener('click', startSpeedTest);
        stopBtn.addEventListener('click', stopSpeedTest);
        closeBtn.addEventListener('click', closeSpeedTestBoard);

        // 点击遮罩层关闭看板
        const boardContainer = document.getElementById('speedtest-board-container');
        if (boardContainer) {
            boardContainer.addEventListener('click', (e) => {
                if (e.target === boardContainer) {
                    closeSpeedTestBoard();
                }
            });
        }

        // 初始化服务器列表
        initServerList();
    };

    // 开始初始化
    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', initSpeedTestPlugin);
    } else {
        initSpeedTestPlugin();
    }
})();
</script>
