                return false;
            } finally {
                state.patcherInstalling = null;
            }
        })();

            refreshStatusUI();

    };

    const doExport = async () => {

        setJob('正在检测设备环境...');

    const doImport = async (file) => {

        setJob('正在检测设备环境...');

            const imgPath = UPLOADS + '/' + uploaded;
            setJob('正在校验上传文件...');
            const remote = await computeChecksumSize(imgPath);
            if (remote.checksum === '__NO_CHECKSUM__' || !remote.checksum) throw new Error('设备侧缺少 SHA-256 命令，已阻止导入');
            if (remote.checksum !== localSum || remote.size !== localSize) throw new Error('上传文件 hash 或大小不一致，已阻止导入');

                title: '确认导入基带',
                danger: true,

分区：<code>${esc(state.partition.name)}</code>（槽位 ${esc(state.slot)}）<br>
by-name：<code>${esc(state.partition.linkPath)}</code><br>
块设备：<code>${esc(state.partition.realPath || state.partition.linkPath)}</code><br>
镜像：<code>${esc(uploaded)}</code><br>
SHA-256：<code>${esc(localSum)}</code>`,

            setJob('空闲');
            refreshStatusUI();

            let patchBodyHtml = `<div style="font-size:.64rem;line-height:1.7;margin-bottom:8px;padding:8px;border-radius:8px;background:rgba(255,255,255,.04);border:1px solid rgba(255,255,255,.08);">
<b>环境检测</b><br>
Root 环境：${avb.hasRoot ? '<span style="color:#4ade80;">已检测到</span>' : '<span style="color:#fca5a5;">未检测到</span>'}<br>
AVB device_state：<code>${esc(avb.deviceState || '未知')}</code><br>
AVB verifiedbootstate：<code>${esc(avb.bootState || '未知')}</code><br>
vbmeta_modem 分区：${avb.hasVbmetaModem ? '<span style="color:#fbbf24;">存在（modem 受 AVB 保护）</span>' : '<span style="color:#4ade80;">不存在（modem 不在验证链中）</span>'}
<div style="margin-top:4px;font-size:.58rem;color:#9aa4b8;">以上信息仅供参考，可能不反映 modem 分区实际保护状态。</div>
</div>`;

                patchBodyHtml += `<div style="margin-bottom:8px;padding:8px;border-radius:8px;background:rgba(239,68,68,.12);border:1px solid rgba(239,68,68,.3);color:#fca5a5;font-weight:700;line-height:1.6;">
⚠️ 检测到 modem 分区在 AVB 验证链中！直接修补基带有几率开机无限重启，请确保设备已禁用 AVB 校验后再操作。</div>`;
            }

            patchBodyHtml += `分区：<code>${esc(state.partition.name || '未知')}</code>（槽位 ${esc(state.slot)}）<br>
by-name：<code>${esc(state.partition.linkPath || '读取失败')}</code><br>
块设备：<code>${esc(state.partition.realPath || state.partition.linkPath || '读取失败')}</code><br><br>
接下来会尝试修补基带，未备份时请务必不要继续！
<div style="margin-top:10px;color:#fca5a5;line-height:1.7;">
⚠️ 新版u20（wifi6）原厂自带显串，无需修补基带！<br>
⚠️ 请务必提前保存旧串号和备份基带，防止无法复原！<br>
⚠️ 强烈建议插入一张普通 SIM 卡并切换到该卡槽，再进行基带修补，否则内置卡可能因串号变更被运营商注销报废！
</div>`;

            const backupOk = await confirmDialog({
                title: '修补前确认',
                danger: modemAvbProtected,
                seconds: modemAvbProtected ? 10 : 8,
                checkbox: true,
                confirmText: '已确认，继续',
                bodyHtml: patchBodyHtml,
            });
            if (!backupOk) { createToast('已取消修补', 'pink', 2000); return; }

            setBusy(true);
            setJob('正在准备 patcher...');
            const prep = await sh(prepareCmd(), SHELL_TO);
            if (!(prep.ok && prep.text.includes('__OK__'))) {
                appendLog(`patcher 准备原始返回：${JSON.stringify(prep.text || prep.raw || '')}`);
                throw new Error('patcher 准备失败：' + explainStageFailure(prep.text, '准备命令未完成'));
            }

            setJob('正在提取并修补 modem...');
            const part = getPartPath();
            const pr = await sh(patchCmd(part), DD_TO);
            if (pr.text) appendLog('patcher 输出：\n' + pr.text.replace(/__OK__\s*$/, '').trim());
            if (!(pr.ok && pr.text.includes('__OK__'))) {
                if (/No changes needed|already patched/i.test(pr.text)) {
                    throw new Error('未产出修补后的镜像，可能目标基带已经修补过！');
                }
                if (/ambiguous or missing|not matched|old=0, ?new=0/i.test(pr.text)) {
                    throw new Error('基带已被修改过，本工具无法再次修补，可尝试直接改串。');

            }
            appendLog(classifyPatch(pr.text));

            setJob('正在计算 patched 镜像校验...');
            const info = await computeChecksumSize(OUTPUT);
            if (info.checksum === '__NO_CHECKSUM__' || !info.checksum) throw new Error('当前环境不可用：缺少 SHA-256 能力');

            const flushOk = await confirmDialog({
                title: '确认刷入修补后的基带',
                danger: true,
                seconds: 6,
                confirmText: '确认刷入',
                bodyHtml: `<div style="background:rgba(255,70,70,.12);border:1px solid rgba(255,90,90,.28);border-radius:8px;padding:8px;margin-bottom:8px;">
这是直接覆盖 <code>${esc(state.partition.name)}</code> 分区的高风险操作。<br>
如果当前基带已经修补过，或修补工具没有产出新镜像，本流程会停止，不会强行刷入。
</div>
by-name：<code>${esc(state.partition.linkPath)}</code><br>
块设备：<code>${esc(state.partition.realPath || state.partition.linkPath)}</code><br>
patched 文件：<code>${esc(OUTPUT)}</code><br>
文件大小：<code>${esc(String(info.size))}</code><br>
SHA-256：<code>${esc(info.checksum)}</code>`,

            createToast('修补后的基带已刷入', 'green');
        } catch (e) {
            console.error(e);
            appendLog('修补失败：' + (e.message || e));

    // ─── AT 改串 ────────────────────────────────────────────
    const parseImeiFromAt = (text) => {
        const cleaned = String(text || '').replace(/OK\s*$/i, '').replace(/ERROR.*$/i, '').trim();
        const m = cleaned.match(/(\d{15})/);
        return m ? m[1] : '';
    };
    const parseImeiFromReader = (text, slot) => {
        const map = {};
        const re = /IMEI\s*(\d)\s*[:：]?\s*(\d{15})/gi;
        let m;
        while ((m = re.exec(text || ''))) map[m[1]] = m[2];
        const key = String(Number(slot) + 1);
        return map[key] || '';
    };

    const executeATCommand = async (command, slot = null) => {
        const targetSlot = slot == null ? '0' : String(slot);
        try {
            const commandEnc = encodeURIComponent(command);
            return await (
                await fetch(`${KANO_baseURL}/AT?command=${commandEnc}&slot=${targetSlot}`, {
                    headers: common_headers,
                })
            ).json();
        } catch (e) {
            return null;
        }
    };

    const execAT = async (command, slot) => {
        if (!command) return { ok: false, data: null, error: 'AT 指令为空' };
        try {
            const res = await executeATCommand(command, slot);
            if (!res) return { ok: false, data: null, error: 'AT 请求无响应' };
            if (res.error) return { ok: false, data: null, error: String(res.error) };
            return { ok: true, data: res.result, error: '' };
        } catch (e) {
            return { ok: false, data: null, error: e?.message || String(e) };
        }
    };

    const queryImeiByCgSn = async (slot) => {
        const res = await execAT('AT+CGSN', slot);
        const raw = String(res?.data || '');
        return {
            ok: !!res?.ok,
            raw,
            error: res?.error || '',
            imei: parseImeiFromAt(raw),
        };
    };

    const logQueryResult = (label, slot, result) => {
        const raw = result.raw || result.error || '';
        appendLog(`[${label}] 槽位 ${slot} 当前 IMEI：${result.imei || '未解析'} 原始返回 ok=${!!result.ok}：${JSON.stringify(raw)}`);
    };

    const atQuery = async (strong) => {
        if (state.busy) return;
        setBusy(true);
        setJob(strong ? '正在强力查串...' : '正在查串...');
        const slot = state.atSlot;
        let queryResult = null;
        try {
            if (strong) {
                const r = await run(sq(IMEI_READER), AT_TO);
                const raw = (r && r.content) || '';
                queryResult = {
                    ok: !!(r && r.success),
                    raw,
                    error: '',
                    imei: parseImeiFromReader(raw, slot),
                };
            } else {
                queryResult = await queryImeiByCgSn(slot);
            }
            logQueryResult(strong ? '强力查串' : '普通查串', slot, queryResult);
            const imei = queryResult.imei;
            if (!imei) throw new Error(strong ? '强力查串未解析到 IMEI（请确认已启用高级功能）' : '查串未解析到 IMEI，可尝试强力查串');
            state.currentImei = imei;
            createToast(' 槽位 ' + slot + ' 当前 IMEI：' + imei, 'green');
        } catch (e) {
            console.error(e);
            if (!queryResult) {
                logQueryResult(strong ? '强力查串' : '普通查串', slot, {
                    ok: false,
                    raw: '',
                    error: e.message || String(e),
                    imei: '',
                });
            }
            createToast(e.message || '查串失败', 'red', 4000);
        } finally {
            setBusy(false);
            setJob('空闲');
            refreshStatusUI();
        }
    };

    const atChange = async () => {
        if (state.busy) return;
        const root = document.querySelector('#' + MODAL);
        const raw = (root?.querySelector('#ikc_imei_input')?.value || '').trim();
        if (!raw) return createToast('请输入 IMEI 号码', 'red', 3000);
        if (/[^\d]/.test(raw)) return createToast('IMEI 只能包含数字', 'red', 3000);
        const imei = raw.replace(/[^\d]/g, '');
        const slot = state.atSlot;

        // 15 位：Luhn 校验
        if (imei.length === 15) {
            if (!luhnValid(imei)) {
                const proceed = await confirmDialog({
                    title: 'IMEI Luhn 校验未通过',
                    danger: true,
                    seconds: 3,
                    confirmText: '仍要写入',
                    bodyHtml: `<div style="color:#fca5a5;font-weight:700;margin-bottom:6px;">该 IMEI 未通过 Luhn 校验，可能是无效串号。</div>
写入无效串号可能导致设备无法正常注册网络。<br>
IMEI：<code>${esc(imei)}</code><br>槽位：<code>${esc(slot)}</code>`,
                });
                if (!proceed) return;
            }
        } else {
            // 1-14 位：短号警告
            const proceed = await confirmDialog({
                title: '非标准 IMEI 长度',
                danger: true,
                seconds: 3,
                confirmText: '仍要写入',
                bodyHtml: `<div style="color:#fca5a5;font-weight:700;margin-bottom:6px;">您输入的不是标准 15 位 IMEI（当前 ${imei.length} 位），部分设备可能无法正常识别或上报。</div>
IMEI：<code>${esc(imei)}</code><br>槽位：<code>${esc(slot)}</code>`,
            });
            if (!proceed) return;
        }

        setBusy(true);
        try {
            setJob('正在查询当前串号...');
            let currentImei = '';
            try {
                const currentResult = await queryImeiByCgSn(slot);
                logQueryResult('改串前查串', slot, currentResult);
                currentImei = currentResult.imei;
            } catch (e) { console.error(e); }
            if (currentImei) { state.currentImei = currentImei; refreshStatusUI(); }
            setJob('等待确认...');

            const ok = await confirmDialog({
                title: '确认改串',
                danger: true,
                seconds: 5,
                confirmText: '确认改串',
                bodyHtml: `当前串号：<code>${esc(currentImei || '查询失败')}</code><br>写入新串号：<code>${esc(imei)}</code><br>槽位：<code>${esc(slot)}</code><br>
若基带未修补，写入可能无效。改串完成后建议重启设备生效。
<div style="margin-top:10px;color:#fca5a5;line-height:1.7;">⚠️ 若您的物联卡绑定了原串号，改串后将自动注销报废！请确保已备份或更换未绑定的卡！</div>`,
            });
            if (!ok) { createToast('已取消改串', 'pink', 2000); return; }
            setJob('正在改串...');
            const atCmd = `AT+SPIMEI=${slot},"${imei}"`;
            const r = await execAT(atCmd, slot);
            const out = String(r?.data || '');
            const transportOk = !!r?.ok;
            const explicitError = /\bERROR\b/i.test(out);
            const hasOk = /\bOK\b/i.test(out);
            appendLog(`改串原始返回 success=${transportOk}：${JSON.stringify(out || r?.error || '')}`);
            if (!transportOk || explicitError) {
                throw new Error('改串指令执行失败：' + (out.trim() || r?.error || '无输出'));
            }
            if (!hasOk) {
                appendLog('改串返回未包含 OK，但未检测到 ERROR，继续通过查串结果确认');
            }
            appendLog(`槽位 ${slot} 已下发改串：${imei}`);
            try {
                const verifyResult = await queryImeiByCgSn(slot);
                logQueryResult('改串后查串', slot, verifyResult);
                const verifyImei = verifyResult.imei;
                if (verifyImei) {
                    state.currentImei = verifyImei;
                    refreshStatusUI();
                    if (verifyImei === imei) {
                        appendLog(`改串后已读到目标 IMEI：${verifyImei}`);
                    } else {
                        appendLog(`改串后当前 IMEI：${verifyImei}，可能需要重启后生效`);
                    }
                } else {
                    appendLog('改串后暂未解析到 IMEI，可能需要重启后再查');
                }
            } catch (e) {
                appendLog('改串后查串失败：' + (e.message || e));
            }
            const doReboot = await confirmDialog({
                title: 'IMEI 修改完成',
                danger: true,
                seconds: 3,
                confirmText: '立即重启',
                cancelText: '稍后重启',
                bodyHtml: `IMEI 修改完成，重启后生效。
<div style="margin-top:10px;color:#fca5a5;line-height:1.7;">⚠️ 建议在确认串号修改成功之前不要插卡！请先重启设备，待串号生效后，再插入需要使用的 SIM 卡。</div>`,
            });
            if (doReboot) {
                appendLog('设备正在重启…');
                createToast('设备正在重启…', 'green');
                try { await runShellWithRoot('reboot'); } catch (e) { console.error(e); }
            } else {
                appendLog('请手动重启设备以生效');
            }
        } catch (e) {
            console.error(e);
            appendLog('改串失败：' + (e.message || e));
            createToast(e.message || '改串失败', 'red', 4000);
        } finally {
            setBusy(false);
            setJob('空闲');
            refreshStatusUI();
        }
    };

    const atReboot = async () => {
        if (state.busy) return;
        const ok = await confirmDialog({
            title: '确认重启设备',
            danger: true,
            seconds: 3,
            confirmText: '重启',
            bodyHtml: '设备将立即重启以使基带更改生效，当前网络连接会中断。',
        });
        if (!ok) return;
        setBusy(true);
        setJob('正在重启设备...');
        appendLog('已发送重启指令');
        createToast('设备正在重启…', 'green');
        try { await runShellWithRoot('reboot'); } catch (e) { console.error(e); }
        finally { setBusy(false); setJob('空闲'); }
    };

        setJob('正在安装 patcher...');

            const ok = await installPatcher(false);

            setBusy(false);
            setJob('空闲');
            refreshStatusUI();
        }
    };

    const doUninstallPatcher = async () => {
        if (state.busy) { createToast('正在执行任务，请稍候', 'pink'); return; }
        setBusy(true);
        setJob('正在卸载 patcher...');
        try {
            await run(`rm -f ${UPLOADS}/*.img ${UPLOADS}/patcher* 2>/dev/null; rm -rf ${sq(TMP)} 2>/dev/null`);
            state.patcherReady = false;
            appendLog('patcher 及缓存已卸载');
            createToast('已卸载修补工具并清理缓存', 'green');
            refreshStatusUI();
        } catch (e) {
            console.error(e);
            appendLog('卸载失败：' + (e.message || e));
            createToast('卸载失败：' + (e.message || e), 'red', 4000);
        } finally {
            setBusy(false);
            setJob('空闲');
            refreshStatusUI();
        }
    };

    // ─── bind ───────────────────────────────────────────────
    const bind = () => {
        const root = document.querySelector('#' + MODAL);
        if (!root) return;
        root.querySelector('#ikc_export_btn').onclick = doExport;
        const fileInput = root.querySelector('#ikc_import_file');
        root.querySelector('#ikc_import_btn').onclick = () => {
            if (state.busy) { createToast('正在执行任务，请稍候', 'pink'); return; }
            fileInput.value = '';
            fileInput.click();
        };
        fileInput.onchange = () => {
            const f = fileInput.files && fileInput.files[0];
            if (f) doImport(f);
        };
        root.querySelector('#ikc_patch_btn').onclick = doPatch;
        root.querySelector('#ikc_slot0').onclick = () => { state.atSlot = '0'; state.currentImei = ''; refreshStatusUI(); };
        root.querySelector('#ikc_slot1').onclick = () => { state.atSlot = '1'; state.currentImei = ''; refreshStatusUI(); };
        root.querySelector('#ikc_imei_cur')?.addEventListener('click', () => copyText(state.currentImei));
        root.querySelector('#ikc_query_btn').onclick = () => atQuery(false);
        root.querySelector('#ikc_squery_btn').onclick = () => atQuery(true);
        root.querySelector('#ikc_rand_btn').onclick = () => {
            const inp = root.querySelector('#ikc_imei_input');
            if (inp && !inp.disabled) inp.value = randomImei();
        };
        root.querySelector('#ikc_change_btn').onclick = atChange;
        root.querySelector('#ikc_reboot_btn').onclick = atReboot;
        root.querySelector('#ikc_patcher_btn').onclick = () => state.patcherReady ? doUninstallPatcher() : doInstallPatcher();
        root.querySelector('#ikc_clear_log_btn').onclick = () => {
            state.logs = [];
            renderLogs();
            appendLog('日志已清空');
        };
    };
    // ─── help ───────────────────────────────────────────────
    const HELP_TEXT = '<b>这是什么</b><br>' +
        '· 本插件用来给设备的“基带”解开限制并修改串号（IMEI，相当于设备的“身份证号”）。<br><br>' +
        '<b>怎么用</b><br>' +
        '· <b>一键修补</b>：改串前必须先点它，解开改串限制；不先修补就改，往往改了不生效、设备认不出，只需做一次。<br>' +
        '· <b>导出</b>=备份当前状态（动手前建议先存一份）；<b>导入</b>=还原备份，或刷入自己准备好的固件。<br>' +
        '· 改串步骤：选“卡槽”→点“查串”读出当前串号→填新串号（或点“随机”）→点“改串”→重启后生效（普通查串和强力查串结果一致）<br>' +
        '· 改完、新号还没确认生效前，先别插绑定原串号的物联卡，以免用不了。<br>' +
        '· 万一改完反复重启开不了机，用之前“导出”的备份，借助其他工具还原即可。<br><br>' +
        '<div style="margin-top:6px;padding:6px 8px;border-radius:8px;background:rgba(239,68,68,.12);border:1px solid rgba(239,68,68,.25);font-size:.6rem;color:#fca5a5;line-height:1.6;">' +
        '<b>免责声明</b><br>' +
        '修补基带、改串号属高风险操作，请在符合当地法律法规前提下使用；因个人使用导致的设备损坏、串号异常、被运营商注销等一切后果，由使用者自行承担。' +
        '</div>';

    const showHelp = () => {
        const { el, close } = createFixedToast('ikc_help_toast', `
            <div style="pointer-events:all;width:80vw;max-width:300px">
                <div class="title" style="margin:0">使用说明</div>
                <div style="margin:10px 0;font-size:.68rem;line-height:1.6">${HELP_TEXT}</div>
                <div style="text-align:right">
                    <button style="font-size:.64rem" id="ikc_help_dismiss">关闭</button>
                </div>
            </div>`);
        el.querySelector('#ikc_help_dismiss').onclick = () => close();
    };

    const injectHelpButton = (modalEl) => {
        const titleSpan = modalEl.querySelector('.title > span');
        if (!titleSpan) return;
        const helpBtn = document.createElement('button');
        helpBtn.textContent = '?';
        helpBtn.style.cssText = 'width:16px;height:16px;border-radius:50%;padding:0;font-size:.5rem;line-height:16px;text-align:center;cursor:pointer;border:1px solid rgba(255,255,255,.15);background:rgba(255,255,255,.08);color:rgba(255,255,255,.7);margin-left:8px;vertical-align:middle;flex-shrink:0;';
        helpBtn.onclick = (e) => { e.stopPropagation(); showHelp(); };
        titleSpan.appendChild(helpBtn);
    };

    // ─── mount ──────────────────────────────────────────────
    const openModal = async () => {
        injectStyle();
        const { close: closeLoading } = createFixedToast('ikc_loading', '初始化中...');
        document.querySelector('#' + MODAL)?.remove();
        try {
            const { el, id } = createModal({
                name: MODAL,
                title: '一键修补基带+改串',
                content: render(),
                showConfirm: false,
                maxWidth: '420px',
                contentStyle: 'max-height: 72vh;',
                onClose: () => {
                    if (state.busy) { createToast('操作进行中，请等待完成后再关闭', 'pink'); return false; }
                    return true;
                },
            });
            showModal(id);
            bind();
            renderLogs();
            updateDisabled();  // bind 后立即刷新按钮状态，state 未就绪时按钮禁用
            injectHelpButton(el);
            refresh();
        } finally {
            closeLoading();
        }
    };

    const btn = document.createElement('button');
    btn.id = ENTRY_ID;
    btn.textContent = '一键修补基带+改串';
    btn.onclick = async () => {
        const who = await run('whoami', 3000);
        if (!((who && who.content) || '').toLowerCase().includes('root')) {
            createToast('请先启用高级功能', 'red');
            return;
        }
        openModal();
    };

    let tries = 0;
    const mountBtn = () => {
        const c = document.querySelector('.actions-buttons');
        if (c) { if (!c.querySelector('#' + ENTRY_ID)) c.appendChild(btn); return; }
        if (tries++ < 40) setTimeout(mountBtn, 150);
    };
    mountBtn();
    } catch (e) {
        console.error('[ikc] unexpected error', e);
        try { createToast && createToast('插件异常：' + (e && e.message || e), 'red', 5000); } catch (_) {}
    }
})();
