            if (!history || history.length < 8) return "检测中";

            const changes = [];
            for (let i = 1; i < history.length; i++) {
                const prev = parseInt(history[i - 1]);
                const curr = parseInt(history[i]);
                if (Math.abs(curr - prev) < 20) {
                    changes.push(Math.abs(curr - prev));
                }
            }

            if (changes.length === 0) return "检测中";

            const avgChange = changes.reduce((a, b) => a + b) / changes.length;

            const thresholds = {
                power: { excellent: 1, good: 3, fair: 6 },
                sinr: { excellent: 0.5, good: 1.5, fair: 3 },
                rsrq: { excellent: 0.3, good: 0.8, fair: 1.5 }
            };

            const threshold = thresholds[type] || thresholds.power;

            if (avgChange < threshold.excellent) return "极稳";
            if (avgChange < threshold.good) return "稳定";
            if (avgChange < threshold.fair) return "一般";
            return "波动";
        }

        // 改进的平均值计算
        function calculateAverage(history, currentValue) {
            const newValue = parseInt(currentValue);

            if (history.length > 0) {
                const currentAvg = history.reduce((a, b) => a + b) / history.length;
                if (Math.abs(newValue - currentAvg) > 15) {
                    history.push((currentAvg * 0.7 + newValue * 0.3));
                } else {
                    history.push(newValue);
                }
            } else {
                history.push(newValue);
            }

            if (history.length > 15) {
                history.shift();
            }

            if (history.length > 0) {
                let weightedSum = 0;
                let weightSum = 0;

                history.forEach((value, index) => {
                    const weight = (index + 1) / history.length;
                    weightedSum += value * weight;
                    weightSum += weight;
                });

                return weightedSum / weightSum;
            }
            return newValue;
        }

        // 计算综合评分
        function calculateOverallScore(data) {
            if (!data.receivePower || !data.sinr || !data.rsrq) {
                return { score: 0, description: "数据不足" };
            }

            const power = parseInt(data.receivePower);
            const sinr = parseInt(data.sinr);
            const rsrq = parseInt(data.rsrq);

            // 计算综合评分 (0-100分)
            let score = 0;

            // RSRP评分 (0-40分)
            if (power >= -83) score += 40;
            else if (power >= -88) score += 35;
            else if (power >= -93) score += 30;
            else if (power >= -98) score += 25;
            else if (power >= -103) score += 20;
            else if (power >= -108) score += 15;
            else if (power >= -113) score += 10;
            else score += 5;

            // SINR评分 (0-35分)
            if (sinr > 25) score += 35;
            else if (sinr > 22) score += 30;
            else if (sinr > 18) score += 25;
            else if (sinr > 14) score += 20;
            else if (sinr > 11) score += 15;
            else if (sinr > 8) score += 10;
            else if (sinr > 6) score += 8;
            else score += 5;

            // RSRQ评分 (0-25分)
            if (rsrq >= -6) score += 25;
            else if (rsrq >= -7) score += 20;
            else if (rsrq >= -8) score += 15;
            else if (rsrq >= -9) score += 12;
            else if (rsrq >= -10) score += 10;
            else if (rsrq >= -11) score += 8;
            else if (rsrq >= -12) score += 6;
            else score += 4;

            // 确定评分描述
            let description;
            if (score >= 90) {
                description = "信号极佳，网络体验优秀";
            } else if (score >= 80) {
                description = "信号良好，网络体验流畅";
            } else if (score >= 70) {
                description = "信号一般，网络体验尚可";
            } else if (score >= 60) {
                description = "信号较差，网络体验一般";
            } else if (score >= 50) {
                description = "信号差，网络体验不佳";
            } else {
                description = "信号极差，建议调整位置";
            }

            return { score, description };
        }

        // 计算趋势
        function calculateTrends(data) {
            if (!data.receivePower || !data.sinr || !data.rsrq) return;

            const currentPower = parseInt(data.receivePower);
            const currentSinr = parseInt(data.sinr);
            const currentRsrq = parseInt(data.rsrq);

            // 计算RSRP趋势
            if (state.previousValues.power !== null) {
                const diff = currentPower - state.previousValues.power;
                if (diff > 2) state.trends.power = 'improving';
                else if (diff < -2) state.trends.power = 'deteriorating';
                else state.trends.power = 'stable';
            }

            // 计算SINR趋势
            if (state.previousValues.sinr !== null) {
                const diff = currentSinr - state.previousValues.sinr;
                if (diff > 1) state.trends.sinr = 'improving';
                else if (diff < -1) state.trends.sinr = 'deteriorating';
                else state.trends.sinr = 'stable';
            }

            // 计算RSRQ趋势
            if (state.previousValues.rsrq !== null) {
                const diff = currentRsrq - state.previousValues.rsrq;
                if (diff > 0.5) state.trends.rsrq = 'improving';
                else if (diff < -0.5) state.trends.rsrq = 'deteriorating';
                else state.trends.rsrq = 'stable';
            }

            // 保存当前值为下一次比较做准备
            state.previousValues.power = currentPower;
            state.previousValues.sinr = currentSinr;
            state.previousValues.rsrq = currentRsrq;

            // 更新显示
            updateTrendDisplay();
        }

        // 绘制信号图表
        function drawSignalChart() {
            const canvas = document.getElementById('signal_chart');
            if (!canvas) return;

            const ctx = canvas.getContext('2d');

            // === 💡 同步Canvas分辨率与显示尺寸 ===
            const dpr = window.devicePixelRatio || 1;
            const displayWidth = canvas.clientWidth;
            const displayHeight = canvas.clientHeight;
            if (canvas.width !== displayWidth * dpr || canvas.height !== displayHeight * dpr) {
                canvas.width = displayWidth * dpr;
                canvas.height = displayHeight * dpr;
            }

            // 缩放绘制坐标到逻辑像素
            ctx.setTransform(dpr, 0, 0, dpr, 0, 0);

            const width = displayWidth;
            const height = displayHeight;

            ctx.clearRect(0, 0, width, height);

            if (!state.chartData || state.chartData.power.length === 0) return;

            const powerData = state.chartData.power;
            const sinrData = state.chartData.sinr;
            const rsrqData = state.chartData.rsrq;

            let datasets = [];
            if (state.chartView === 'all' || state.chartView === 'power') {
                datasets.push({ data: powerData, color: '#52ef58', label: 'RSRP' });
            }
            if (state.chartView === 'all' || state.chartView === 'sinr') {
                datasets.push({ data: sinrData, color: '#2196F3', label: 'SINR' });
            }
            if (state.chartView === 'all' || state.chartView === 'rsrq') {
                datasets.push({ data: rsrqData, color: '#FF9800', label: 'RSRQ' });
            }

            // 计算数据范围
            let minValue = Infinity;
            let maxValue = -Infinity;
            datasets.forEach(ds => {
                ds.data.forEach(v => {
                    if (v < minValue) minValue = v;
                    if (v > maxValue) maxValue = v;
                });
            });

            const range = maxValue - minValue;
            minValue = minValue - range * 0.1;
            maxValue = maxValue + range * 0.1;
            if (minValue === maxValue) {
                minValue -= 1;
                maxValue += 1;
            }

            // === 绘制网格线 ===
            ctx.strokeStyle = 'rgba(255,255,255,0.1)';
            ctx.lineWidth = 0.5;
            for (let i = 0; i <= 4; i++) {
                const y = height - (i * height / 4);
                ctx.beginPath();
                ctx.moveTo(0, y);
                ctx.lineTo(width, y);
                ctx.stroke();
            }

            // 按显示层级绘制
            const orderedDatasets = [];
            if (state.chartView === 'all' || state.chartView === 'rsrq')
                orderedDatasets.push({ data: rsrqData, color: '#FF9800', label: 'RSRQ' });
            if (state.chartView === 'all' || state.chartView === 'sinr')
                orderedDatasets.push({ data: sinrData, color: '#2196F3', label: 'SINR' });
            if (state.chartView === 'all' || state.chartView === 'power')
                orderedDatasets.push({ data: powerData, color: '#52ef58', label: 'RSRP' });

            orderedDatasets.forEach(ds => {
                const data = ds.data;
                if (data.length < 2) return;

                ctx.strokeStyle = ds.color;
                ctx.lineWidth = 1.5;
                ctx.beginPath();

                data.forEach((v, i) => {
                    const x = (i / (data.length - 1)) * width;
                    const y = height - ((v - minValue) / (maxValue - minValue)) * height;
                    if (i === 0) ctx.moveTo(x, y);
                    else ctx.lineTo(x, y);
                });

                ctx.stroke();

                // 绘制数据点
                ctx.fillStyle = ds.color;
                data.forEach((v, i) => {
                    const x = (i / (data.length - 1)) * width;
                    const y = height - ((v - minValue) / (maxValue - minValue)) * height;
                    ctx.beginPath();
                    ctx.arc(x, y, 2, 0, Math.PI * 2);
                    ctx.fill();

                    if (i === data.length - 1) {
                        ctx.fillStyle = 'var(--dark-text-color)';
                        ctx.font = '8px Arial';
                        ctx.textAlign = 'center';
                        ctx.fillText(v, x - 10, y - 8);
                    }
                });
            });
        }

        // 改进的数据解析
        async function parseSignalFromPage() {
            return new Promise((resolve) => {
                try {
                    setTimeout(() => {
                        try {
                            const pageText = document.body.innerText;
                            let signalData = null;

                            const patterns = [
                                {
                                    power: pageText.match(/5G接收功率[：:\s]*(-?\d+)/),
                                    band: pageText.match(/5G注册频段[：:\s]*([^\s]+)/),
                                    sinr: pageText.match(/5G SINR[：:\s]*(-?\d+)/),
                                    rsrq: pageText.match(/5G RSRQ[：:\s]*(-?\d+)/),
                                    pci: pageText.match(/5G PCI[：:\s]*(\d+)/)
                                },
                                {
                                    power: pageText.match(/RSRP[：:\s]*(-?\d+)/),
                                    band: pageText.match(/频段[：:\s]*([^\s]+)/),
                                    sinr: pageText.match(/SINR[：:\s]*(-?\d+)/),
                                    rsrq: pageText.match(/RSRQ[：:\s]*(-?\d+)/),
                                    pci: pageText.match(/PCI[：:\s]*(\d+)/)
                                },
                                {
                                    power: pageText.match(/RSRP[:\s]*(-?\d+)/),
                                    band: pageText.match(/Band[:\s]*([^\s]+)/),
                                    sinr: pageText.match(/SINR[:\s]*(-?\d+)/),
                                    rsrq: pageText.match(/RSRQ[:\s]*(-?\d+)/),
                                    pci: pageText.match(/PCI[:\s]*(\d+)/)
                                }
                            ];

                            for (const pattern of patterns) {
                                const validCount = Object.values(pattern).filter(match => match !== null).length;
                                if (validCount >= 3) {
                                    signalData = {
                                        receivePower: pattern.power ? pattern.power[1] : null,
                                        band: pattern.band ? pattern.band[1] : null,
                                        sinr: pattern.sinr ? pattern.sinr[1] : null,
                                        rsrq: pattern.rsrq ? pattern.rsrq[1] : null,
                                        pci: pattern.pci ? pattern.pci[1] : null
                                    };
                                    break;
                                }
                            }

                            if (signalData) {
                                if (signalData.receivePower && (signalData.receivePower > -50 || signalData.receivePower < -150)) {
                                    signalData.receivePower = null;
                                }
                                if (signalData.sinr && (signalData.sinr > 50 || signalData.sinr < -10)) {
                                    signalData.sinr = null;
                                }
                                if (signalData.rsrq && (signalData.rsrq > 0 || signalData.rsrq < -30)) {
                                    signalData.rsrq = null;
                                }
                            }

                            const hasData = signalData && Object.values(signalData).some(value => value !== null);
                            resolve(hasData ? signalData : null);
                        } catch (parseError) {
                            console.warn('数据解析错误:', parseError);
                            resolve(null);
                        }
                    }, 150);
                } catch (error) {
                    resolve(null);
                }
            });
        }

        // 获取信号数据
        async function fetchSignalData() {
            try {
                let signalData = await parseSignalFromPage();
                let isSimulated = false;

                if (!signalData) {
                    isSimulated = true;
                    const basePower = state.history.power.length > 0 ?
                        state.history.power[state.history.power.length - 1] : -85;
                    const baseSinr = state.history.sinr.length > 0 ?
                        state.history.sinr[state.history.sinr.length - 1] : 20;
                    const baseRsrq = state.history.rsrq.length > 0 ?
                        state.history.rsrq[state.history.rsrq.length - 1] : -8;

                    signalData = {
                        receivePower: (basePower - 2 + Math.random() * 4).toFixed(0),
                        band: 'N78',
                        sinr: (baseSinr - 1 + Math.random() * 2).toFixed(0),
                        rsrq: (baseRsrq - 0.5 + Math.random() * 1).toFixed(0),
                        pci: '313'
                    };
                }

                // 更新最佳值记录
                updateBestValues(signalData);

                // 更新图表数据
                updateChartData(signalData);

                // 更新当前数值显示
                updateCurrentValues(signalData);

                // 计算趋势
                calculateTrends(signalData);

                // 计算综合评分
                const scoreData = calculateOverallScore(signalData);
                document.getElementById('overall_score').textContent = scoreData.score;
                document.getElementById('score_description').textContent = scoreData.description;

                state.dataSource = isSimulated ? 'simulated' : 'parsing';
                state.sampleCount++;
                state.lastDataTime = new Date();

                updateSignalDisplay(signalData);

            } catch (error) {
                console.error('获取信号数据错误:', error);
            }
        }

        // 获取当前更新间隔
        function getCurrentInterval() {
            const select = document.getElementById('update_interval_select');
            return parseInt(select.value);
        }

        // 重启监控（当间隔改变时调用）
        function restartMonitoring() {
            if (state.isMonitoring) {
                clearInterval(state.monitorInterval);
                state.monitorInterval = setInterval(fetchSignalData, getCurrentInterval());
            }
        }

        // 切换自动监控
        function toggleAutoMonitor() {
            if (state.isMonitoring) {
                clearInterval(state.monitorInterval);
                document.getElementById('auto_monitor_btn').textContent = '开始监控';
                document.getElementById('auto_monitor_btn').style.background = '#666';
                state.isMonitoring = false;
            } else {
                state.monitorInterval = setInterval(fetchSignalData, getCurrentInterval());
                document.getElementById('auto_monitor_btn').textContent = '监控中';
                document.getElementById('auto_monitor_btn').style.background = '#52ef58';
                state.isMonitoring = true;
            }
        }

        // 初始化函数
        function initSignalMonitor() {
            // 初始化状态点
            initializeStatusDots();

            // 初始化图表视图按钮
            document.getElementById('toggle_chart_btn').textContent = '视图: 全部';

            // 初始化趋势显示
            updateTrendDisplay();

            // 绑定事件

            document.getElementById('toggle_chart_btn').onclick = toggleChartView;
            document.getElementById('toggle_chart_display_btn').onclick = toggleChartDisplay;

        }
