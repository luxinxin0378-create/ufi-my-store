  class Carousel {
      constructor(viewport, track, dotsContainer, isNested = false, onIndexChange = null, initialIndex = 0) {
          this.viewport = viewport; this.track = track; this.dotsContainer = dotsContainer; 
          this.isNested = isNested; this.onIndexChange = onIndexChange;
          this.index = initialIndex; this.startX = 0; this.isDragging = false; 
          this.init();
          if(this.index > 0) this.slideTo(this.index, false); 
      }
      updateDots(count) {
          if(!this.dotsContainer) return; this.dotsContainer.innerHTML = ''; if(count <= 1) return; 
          for(let i=0; i<count; i++) {
              const dot = document.createElement('div');
              dot.className = `tm-dot ${i === this.index ? 'active' : ''}`;
              dot.onclick = (e) => { e.stopPropagation(); this.slideTo(i); };
              this.dotsContainer.appendChild(dot);
          }
      }
      slideTo(index, animate = true) {
          const maxIndex = this.track.children.length - 1;
          if (index < 0) index = 0; if (index > maxIndex) index = maxIndex;
          this.index = index;
          if(this.onIndexChange) this.onIndexChange(index);
          this.track.style.transition = animate ? 'transform 0.4s cubic-bezier(0.25, 1, 0.5, 1)' : 'none';
          this.track.style.transform = `translateX(-${index * 100}%)`;
          this.updateDots(this.track.children.length);
      }
      init() {
          let isMoved = false;
          const handleStart = (e) => { if(this.track.children.length <= 1) return; if(this.isNested) e.stopPropagation(); this.isDragging = true; isMoved = false; this.startX = e.type.includes('mouse') ? e.clientX : e.touches[0].clientX; this.track.style.transition = 'none'; };
          const handleMove = (e) => { if (!this.isDragging) return; if(this.isNested) e.stopPropagation(); const currentX = e.type.includes('mouse') ? e.clientX : e.touches[0].clientX; const diffX = currentX - this.startX; if (Math.abs(diffX) < 5) return; isMoved = true; if (e.cancelable) e.preventDefault(); this.track.style.transform = `translateX(${-(this.index * this.viewport.offsetWidth) + diffX}px)`; };
          const handleEnd = (e) => { if (!this.isDragging) return; if(this.isNested) e.stopPropagation(); this.isDragging = false; if (!isMoved) return; const endX = e.type.includes('mouse') ? e.clientX : e.changedTouches[0].clientX; const diff = endX - this.startX; if (Math.abs(diff) > 50) this.slideTo(diff > 0 ? this.index - 1 : this.index + 1); else this.slideTo(this.index); };
          this.viewport.addEventListener('touchstart', handleStart, { passive: true }); this.viewport.addEventListener('touchmove', handleMove, { passive: false }); this.viewport.addEventListener('touchend', handleEnd); this.viewport.addEventListener('mousedown', handleStart); this.viewport.addEventListener('mousemove', handleMove); this.viewport.addEventListener('mouseup', handleEnd); this.viewport.addEventListener('mouseleave', handleEnd);
          setTimeout(() => this.updateDots(this.track.children.length), 100);
      }
  }

  window.animateValue = function(id, start, end, duration) {
      const obj = document.getElementById(id); if (!obj) return;
      if (isNaN(parseFloat(start)) || isNaN(parseFloat(end))) { obj.innerHTML = end; return; }
      if (AnimationRegistry[id]) cancelAnimationFrame(AnimationRegistry[id]);
      
      let startTimestamp = null;
      const step = (timestamp) => {
          if (!startTimestamp) startTimestamp = timestamp;
          const progress = Math.min((timestamp - startTimestamp) / duration, 1);
          const ease = 1 - Math.pow(1 - progress, 4);
          const current = parseFloat(start) + (parseFloat(end) - parseFloat(start)) * ease;
          
          if (id === 'dashboard_status') { obj.innerHTML = Math.floor(current); } 
          else { obj.innerHTML = current.toFixed(id === 'dashboard_percent' ? 0 : 2); }
          
          if (progress < 1) { 
              AnimationRegistry[id] = window.requestAnimationFrame(step); 
          } else { 
            obj.innerHTML = id === 'dashboard_percent' ? end.toFixed(0) : (id === 'dashboard_status' ? end : parseFloat(end).toFixed(2)); 
          }
      };
      AnimationRegistry[id] = window.requestAnimationFrame(step);
  };

  function enableMenuSorting() {
      const container = document.getElementById('tm_menu_overlay');
      let draggedItem = null;
      let touchStartY = 0;
      let dragSrcEl = null;

      container.querySelectorAll('.tm-menu-item').forEach(item => {
          item.setAttribute('draggable', 'true');
          item.addEventListener('dragstart', (e) => {
              draggedItem = item;
              e.dataTransfer.effectAllowed = 'move';
              e.dataTransfer.setData('text/html', item.innerHTML);
              setTimeout(() => item.classList.add('dragging'), 0);
          });
          item.addEventListener('dragend', () => {
              draggedItem = null;
              item.classList.remove('dragging');
              saveSortOrder();
          });
          item.addEventListener('dragover', (e) => {
              e.preventDefault(); 
              e.dataTransfer.dropEffect = 'move';
              const afterElement = getDragAfterElement(container, e.clientY);
              if (afterElement == null) {
                  container.appendChild(draggedItem);
              } else {
                  container.insertBefore(draggedItem, afterElement);
              }
          });

          item.addEventListener('touchstart', (e) => {
              if(e.target.closest('.tm-menu-handle') || e.target.classList.contains('tm-menu-item')) {
                  touchStartY = e.touches[0].clientY;
                  dragSrcEl = item;
                  setTimeout(() => item.classList.add('dragging'), 200);
              }
          }, {passive: false});

          item.addEventListener('touchmove', (e) => {
              if (!dragSrcEl) return;
              e.preventDefault();
              const touchY = e.touches[0].clientY;
              const afterElement = getDragAfterElement(container, touchY);
              if (afterElement) {
                  container.insertBefore(dragSrcEl, afterElement);
              } else {
                  container.appendChild(dragSrcEl);
              }
          }, {passive: false});

          item.addEventListener('touchend', () => {
              if (dragSrcEl) {
                  dragSrcEl.classList.remove('dragging');
                  dragSrcEl = null;
                  saveSortOrder();
              }
          });
      });
  }

  function getDragAfterElement(container, y) {
      const draggableElements = [...container.querySelectorAll('.tm-menu-item:not(.dragging)')];
      return draggableElements.reduce((closest, child) => {
          const box = child.getBoundingClientRect();
          const offset = y - box.top - box.height / 2;
          if (offset < 0 && offset > closest.offset) {
              return { offset: offset, element: child };
          } else {
              return closest;
          }
      }, { offset: Number.NEGATIVE_INFINITY }).element;
  }

  function updateDashboardUI() {
      const keys = Object.keys(GlobalDataCache);
      let totalGB = 0, usedGB = 0;
      keys.forEach(k => { const i = GlobalDataCache[k]; if(i && !i.error) { totalGB += parseFloat(i.total)||0; usedGB += parseFloat(i.used)||0; } });
      const remainGB = totalGB - usedGB;
      const percent = totalGB > 0 ? ((usedGB/totalGB)*100).toFixed(0) : 0;
      
      const today = new Date().getDate();
      let estimatedDaysStr = '--';
      if (usedGB > 0 && today > 0) { const avg = usedGB / today; if (avg > 0) estimatedDaysStr = Math.floor((totalGB - usedGB) / avg); if (estimatedDaysStr > 999) estimatedDaysStr = '999+'; } else if (totalGB - usedGB > 0) estimatedDaysStr = '30+';
