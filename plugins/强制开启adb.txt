<script>
    (() => {
        const menuBox = collapseBtn_menu.nextElementSibling.querySelector('.collapse_box')

        const btnWirelessAdb = document.createElement('button')
        btnWirelessAdb.textContent = "开启ADB"
        menuBox.appendChild(btnWirelessAdb)

        btnWirelessAdb.onclick = async () => {
            try {
                createToast("正在开启无线 ADB...", "")
                await runShellWithRoot("setprop service.adb.tcp.port 5555")
                createToast("正在开启有线 ADB...", "")
                const mode = (await runShellWithRoot("getprop sys.usb.config")).content.trim().replace('\n', '')
                if (mode.includes("adb")) {
                    createToast("有线 ADB 已开启，无需重复开启")
                    return
                } else {
                    await runShellWithRoot(`setprop sys.usb.config ${mode},adb`)
                }
                await runShellWithRoot("stop adbd")
                await runShellWithRoot("start adbd")
                createToast(`✅ 无线+有线ADB已开启：adb connect ip:5555`, "green")
            } catch (e) {
                console.error(e)
                createToast("执行失败，可能无高级功能", "red")
            }
        }
    })();
</script>
